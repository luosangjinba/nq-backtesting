// Get chart time for a bar (daily uses tradingDay, intraday uses timestamp)
export function getBarChartTime(bar, timeframe) {
  if (!bar) return null;
  return Number(timeframe) === 1440 ? bar.tradingDay : bar.timestamp;
}
